Zarezerwuj wizytę

Jakub Banasiewicz

@startuml start :Wybierz opcje "Zarezerwuj wizyte"; repeat :Wyswietl liste pojazdow klienta; backward:Dodaj pojazd (przypadek uzycia "Dodaj pojazd"); repeat while (Klient nie ma pojazdu lub wybiera "Dodaj nowy pojazd"?) is (tak) not (nie) :Wybierz pojazd z listy; if (Pojazd ma aktywna rezerwacje lub otwarte zlecenie?) then (tak) :Poinformuj o istniejacej rezerwacji lub zleceniu; else (nie) :Wyswietl kalendarz z wolnymi terminami; :Wybierz termin wizyty; repeat :Wyswietl formularz opisu usterki; :Wpisz opis usterki; if (Klient anuluje?) then (tak) :Nie tworz rezerwacji; break else (nie) endif :Wyslij formularz; :Zwaliduj formularz; if (Opis usterki pusty?) then (tak) :Zaznacz brakujace pole; else (nie) :Sprawdz, czy termin jest nadal wolny; if (Termin zajety w miedzyczasie?) then (tak) :Poinformuj, ze termin jest niedostepny; :Wyswietl ponownie kalendarz z wolnymi terminami; :Wybierz inny termin wizyty; else (nie) :Zapisz rezerwacje "oczekuje na potwierdzenie" i zablokuj termin; :Przekaz rezerwacje managerowi do potwierdzenia; :Wyswietl potwierdzenie zlozenia rezerwacji; break endif endif repeat while endif stop @enduml
