Budowniczy

Łukasz Bera

Problem

W przypadku użycia „Zamów części u magazyniera” mechanik składa zamówienie stopniowo: przegląda katalog, wybiera kolejne części i podaje ich ilości (krok 7), a dopiero potem wysyła zamówienie (krok 8). System waliduje je w kroku 9 i odrzuca, gdy nie wybrano żadnej części, ilość jest niepoprawna albo ta sama część została już zamówiona do tego zlecenia i nie została jeszcze wydana (przepływy alternatywne 9, GOTO 6). Dopiero poprawne zamówienie jest tworzone, przypisywane do zlecenia i przekazywane magazynierowi (kroki 10–11).

Gdyby obiekt PartRequest tworzyć od razu i dopisywać do niego pozycje w trakcie wybierania części, w systemie istniałyby zamówienia niepełne lub błędne, np. bez żadnej pozycji, choć diagram klas wymaga co najmniej jednej (kompozycja PartRequest–PartItem, 1..*). Walidacja byłaby rozproszona po kodzie, a magazynier mógłby zobaczyć zamówienie, którego mechanik jeszcze nie wysłał.

Rozwiązanie

Budowniczy oddziela proces tworzenia złożonego obiektu z części od jego reprezentacji. Obiekt jest składany krok po kroku, a powstaje dopiero w metodzie build(), po sprawdzeniu, że wszystkie jego części są poprawne.

  • PartRequestBuilder – budowniczy. selectPart() odpowiada wyborowi części z katalogu i podaniu ilości (ponowny wybór tej samej części zmienia ilość), removePart() – rezygnacji z części przy poprawianiu zamówienia. build() wykonuje walidację z kroku 9 i tworzy zamówienie.
  • PartRequest – produkt złożony z pozycji PartItem. Jego konstruktor i metoda addItem() mają widoczność pakietową, więc zamówienie może utworzyć tylko budowniczy. Nowe zamówienie ma status PENDING.
  • existingRequests – zamówienia już złożone (np. pobrane przez WarehouseService.findPendingRequests()). Budowniczy sprawdza w nich, czy część nie została już zamówiona do tego samego zlecenia i nie czeka na wydanie (status PENDING, WAITING lub READY_FOR_ISSUE).

Metody budowniczego zwracają this, więc wywołania można łączyć w łańcuch. Po błędzie walidacji budowniczy pamięta wybrane części, więc mechanik poprawia tylko błędną pozycję i wysyła zamówienie ponownie. Odpowiada to powrotowi do katalogu części (GOTO 6). Zastosowano uproszczoną wersję wzorca z jednym budowniczym, tak jak w przykładzie z materiałów. Klasy SparePart i RepairOrder są w kodzie uproszczone do pól potrzebnych w zamówieniu.

import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

// Budowniczy zamówienia części (przypadek użycia "Zamów części u magazyniera")
public class PartRequestBuilder {
    private final RepairOrder repairOrder;
    private final List<PartRequest> existingRequests;   // zamówienia już złożone do zleceń
    private final Map<SparePart, Integer> selectedParts = new LinkedHashMap<>();

    public PartRequestBuilder(RepairOrder repairOrder, List<PartRequest> existingRequests) {
        this.repairOrder = repairOrder;
        this.existingRequests = existingRequests;
    }

    // Krok 7: mechanik wybiera część z katalogu i podaje ilość.
    // Ponowny wybór tej samej części zmienia jej ilość.
    public PartRequestBuilder selectPart(SparePart part, int quantity) {
        selectedParts.put(part, quantity);
        return this;
    }

    // Poprawa zamówienia po błędzie walidacji (GOTO 6) - rezygnacja z wybranej części
    public PartRequestBuilder removePart(SparePart part) {
        selectedParts.remove(part);
        return this;
    }

    // Kroki 8-10: wysłanie zamówienia, walidacja i utworzenie zamówienia przypisanego do zlecenia
    public PartRequest build() {
        validate();
        PartRequest request = new PartRequest(repairOrder);
        for (Map.Entry<SparePart, Integer> entry : selectedParts.entrySet()) {
            request.addItem(entry.getKey(), entry.getValue());
        }
        return request;
    }

    // Krok 9 i przepływy alternatywne z kroku 9
    private void validate() {
        if (selectedParts.isEmpty()) {
            throw new IllegalStateException("Nie wybrano żadnej części");
        }
        for (Map.Entry<SparePart, Integer> entry : selectedParts.entrySet()) {
            SparePart part = entry.getKey();
            if (entry.getValue() <= 0) {
                throw new IllegalStateException(
                        "Niepoprawna ilość części " + part.getName() + ": " + entry.getValue());
            }
            if (isAlreadyOrdered(part)) {
                throw new IllegalStateException("Część " + part.getName()
                        + " została już zamówiona do tego zlecenia i nie została jeszcze wydana");
            }
        }
    }

    private boolean isAlreadyOrdered(SparePart part) {
        for (PartRequest request : existingRequests) {
            if (request.getRepairOrder() == repairOrder && request.isNotIssued() && request.containsPart(part)) {
                return true;
            }
        }
        return false;
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

// Zamówienie części - produkt budowniczego
public class PartRequest {
    private static int nextId = 1;

    private final int id;
    private final RepairOrder repairOrder;
    private final List<PartItem> items = new ArrayList<>();
    private PartRequestStatus status;

    // Konstruktor i addItem są pakietowe - zamówienie tworzy tylko PartRequestBuilder
    PartRequest(RepairOrder repairOrder) {
        this.id = nextId++;
        this.repairOrder = repairOrder;
        this.status = PartRequestStatus.PENDING;
    }

    void addItem(SparePart part, int quantity) {
        items.add(new PartItem(part, quantity));
    }

    public boolean containsPart(SparePart part) {
        for (PartItem item : items) {
            if (item.getPart() == part) {
                return true;
            }
        }
        return false;
    }

    // Zamówienie złożone, ale części jeszcze nie zostały wydane mechanikowi
    public boolean isNotIssued() {
        return status == PartRequestStatus.PENDING
                || status == PartRequestStatus.WAITING
                || status == PartRequestStatus.READY_FOR_ISSUE;
    }

    public void markFulfilled() {
        status = PartRequestStatus.FULFILLED;
    }

    public int getId() { return id; }

    public RepairOrder getRepairOrder() { return repairOrder; }

    public List<PartItem> getItems() { return Collections.unmodifiableList(items); }

    public PartRequestStatus getStatus() { return status; }
}

// Pozycja zamówienia: część i zamawiana ilość
public class PartItem {
    private final SparePart part;
    private final int quantity;

    public PartItem(SparePart part, int quantity) {
        this.part = part;
        this.quantity = quantity;
    }

    public SparePart getPart() { return part; }

    public int getQuantity() { return quantity; }
}

// Część zamienna z katalogu magazynu (uproszczona)
public class SparePart {
    private final String catalogNumber;
    private final String name;

    public SparePart(String catalogNumber, String name) {
        this.catalogNumber = catalogNumber;
        this.name = name;
    }

    public String getName() { return name; }
}

// Zlecenie naprawy (uproszczone - tylko numer zlecenia)
public class RepairOrder {
    private final int id;

    public RepairOrder(int id) {
        this.id = id;
    }

    public int getId() { return id; }
}

public enum PartRequestStatus {
    PENDING,
    WAITING,
    READY_FOR_ISSUE,
    FULFILLED,
    CANCELLED
}
Kod testowy wzorca
import static org.junit.jupiter.api.Assertions.*;

import java.util.ArrayList;
import java.util.List;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

public class PartRequestBuilderTest {
    private RepairOrder order;
    private SparePart oilFilter;
    private SparePart brakePads;
    private List<PartRequest> existingRequests;

    @BeforeEach
    public void setUp() {
        order = new RepairOrder(15);
        oilFilter = new SparePart("FO-101", "Filtr oleju");
        brakePads = new SparePart("KL-10", "Klocki hamulcowe");
        existingRequests = new ArrayList<>();
    }

    @Test
    public void testBuildRequestWithSelectedParts() {
        PartRequest request = new PartRequestBuilder(order, existingRequests)
                .selectPart(oilFilter, 1)
                .selectPart(brakePads, 4)
                .build();

        assertSame(order, request.getRepairOrder());
        assertEquals(PartRequestStatus.PENDING, request.getStatus());
        assertEquals(2, request.getItems().size());
        assertEquals(4, request.getItems().get(1).getQuantity());
    }

    @Test
    public void testSelectingSamePartAgainChangesQuantity() {
        PartRequest request = new PartRequestBuilder(order, existingRequests)
                .selectPart(brakePads, 2)
                .selectPart(brakePads, 4)
                .build();

        assertEquals(1, request.getItems().size());
        assertEquals(4, request.getItems().get(0).getQuantity());
    }

    @Test
    public void testRequestWithoutPartsIsRejected() {
        PartRequestBuilder builder = new PartRequestBuilder(order, existingRequests);

        IllegalStateException e = assertThrows(IllegalStateException.class, builder::build);
        assertEquals("Nie wybrano żadnej części", e.getMessage());
    }

    @Test
    public void testInvalidQuantityIsRejectedAndCanBeCorrected() {
        PartRequestBuilder builder = new PartRequestBuilder(order, existingRequests)
                .selectPart(oilFilter, 0);

        assertThrows(IllegalStateException.class, builder::build);

        // GOTO 6: mechanik poprawia ilość
        PartRequest request = builder.selectPart(oilFilter, 1).build();
        assertEquals(1, request.getItems().get(0).getQuantity());
    }

    @Test
    public void testPartAlreadyOrderedAndNotIssuedIsRejected() {
        existingRequests.add(new PartRequestBuilder(order, existingRequests).selectPart(brakePads, 4).build());

        PartRequestBuilder builder = new PartRequestBuilder(order, existingRequests)
                .selectPart(brakePads, 2)
                .selectPart(oilFilter, 1);

        IllegalStateException e = assertThrows(IllegalStateException.class, builder::build);
        assertTrue(e.getMessage().contains("została już zamówiona"));

        // GOTO 6: mechanik rezygnuje z klocków, zamawia tylko filtr
        PartRequest request = builder.removePart(brakePads).build();
        assertEquals(1, request.getItems().size());
        assertSame(oilFilter, request.getItems().get(0).getPart());
    }

    @Test
    public void testIssuedPartCanBeOrderedAgain() {
        PartRequest issued = new PartRequestBuilder(order, existingRequests).selectPart(brakePads, 4).build();
        issued.markFulfilled();
        existingRequests.add(issued);

        PartRequest request = new PartRequestBuilder(order, existingRequests).selectPart(brakePads, 2).build();

        assertTrue(request.containsPart(brakePads));
    }

    @Test
    public void testSamePartForOtherOrderIsAllowed() {
        existingRequests.add(new PartRequestBuilder(order, existingRequests).selectPart(brakePads, 4).build());

        PartRequest request = new PartRequestBuilder(new RepairOrder(16), existingRequests)
                .selectPart(brakePads, 4)
                .build();

        assertTrue(request.containsPart(brakePads));
    }
}
