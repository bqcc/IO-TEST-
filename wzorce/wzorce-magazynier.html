Wzorce projektowe

Metoda fabrykująca

Kacper Bednarz

Problem

W przypadku użycia „Wydaj części mechanikowi” system po sprawdzeniu dostępności zamówionych części (krok 6) postępuje na jeden z trzech sposobów:

  • wszystkie części są dostępne – wydanie pełne: stan magazynowy jest pomniejszany, a zamówienie oznaczane jako zrealizowane (kroki 9–11);
  • dostępna ilość jest mniejsza niż zamówiona – wydanie częściowe: brakująca ilość jest usuwana z zamówienia, a mechanik musi zamówić ją ponownie (przepływ alternatywny 6, GOTO 7);
  • części nie ma w magazynie – nic nie jest wydawane, zamówienie dostaje status oczekującego, a zlecenie status „oczekuje na części” (przepływ alternatywny 6, END FLOW).

Każdy sposób inaczej tworzy podsumowanie wydania (krok 7) i inaczej zmienia stan systemu. Gdyby WarehouseServiceImpl rozróżniał te przypadki instrukcjami warunkowymi w każdym miejscu, w którym ich potrzebuje (podsumowanie, zapis wydania), logika wydania byłaby rozproszona, a dodanie nowego sposobu wydania wymagałoby zmian w wielu metodach.

Rozwiązanie

Metoda fabrykująca definiuje sposób tworzenia obiektu, ale zwalnia kod klienta z decyzji, jaki konkretny obiekt ma powstać. Tutaj decyzja zależy od stanu magazynu, a nie od kodu, który wydaje części.

  • PartIssue – produkt abstrakcyjny. getSummary() zwraca treść podsumowania wydania, a execute() zapisuje skutki wydania w systemie.
  • FullIssue, PartialIssue, WaitingIssue – produkty konkretne dla trzech sposobów ze scenariusza. Ich konstruktory są pakietowe, więc tworzy je tylko fabryka.
  • PartIssueFactory.createIssue() – metoda fabrykująca. Sprawdza wolną ilość każdej części (stan magazynowy minus ilość zarezerwowana) i zwraca odpowiedni produkt. Brak którejkolwiek części oznacza, że całe zamówienie czeka na dostawę.
  • WarehouseServiceImpl – klient. W metodach getIssueSummary() i issueParts() korzysta tylko z typu PartIssue.

Przepływ alternatywny 9 (zlecenie zakończone lub rozliczone w międzyczasie) dotyczy zarówno wydania pełnego, jak i częściowego, dlatego metoda cancelIfOrderClosed() znajduje się w klasie bazowej. Wydanie częściowe korzysta z metody PartItem.reduceQuantity() z diagramu klas. Struktura odpowiada schematowi metody fabrykującej z instrukcji (fabryka, produkt abstrakcyjny, produkty konkretne). Klasy SparePart, PartRequest i RepairOrder są w kodzie uproszczone do pól potrzebnych przy wydaniu.

import java.util.LinkedHashMap;
import java.util.Map;

// Fabryka z metodą fabrykującą - decyduje, jaki rodzaj wydania utworzyć
public final class PartIssueFactory {

    private PartIssueFactory() {
    }

    // Krok 6: na podstawie dostępności zamówionych części tworzy odpowiedni rodzaj wydania
    public static PartIssue createIssue(PartRequest request) {
        Map<PartItem, Integer> missingQuantities = new LinkedHashMap<>();
        for (PartItem item : request.getItems()) {
            int available = item.getPart().getAvailableQuantity();
            if (available <= 0) {
                return new WaitingIssue(request);   // część niedostępna w magazynie
            }
            if (available < item.getQuantity()) {
                missingQuantities.put(item, item.getQuantity() - available);
            }
        }
        if (missingQuantities.isEmpty()) {
            return new FullIssue(request);
        }
        return new PartialIssue(request, missingQuantities);
    }
}

// Produkt abstrakcyjny - sposób wydania części dla zamówienia
public abstract class PartIssue {
    protected final PartRequest request;

    protected PartIssue(PartRequest request) {
        this.request = request;
    }

    // Krok 7: treść podsumowania wydania pokazywana magazynierowi
    public abstract String getSummary();

    // Skutki wydania w systemie (wywoływane po potwierdzeniu przez magazyniera)
    public abstract void execute();

    // Przepływ alternatywny 9: zlecenie zostało w międzyczasie zakończone lub rozliczone
    protected boolean cancelIfOrderClosed() {
        RepairOrderStatus orderStatus = request.getRepairOrder().getStatus();
        if (orderStatus == RepairOrderStatus.COMPLETED || orderStatus == RepairOrderStatus.SETTLED) {
            request.cancel();   // stan magazynowy pozostaje bez zmian
            return true;
        }
        return false;
    }
}

// Produkt konkretny - wydanie wszystkich zamówionych części
public class FullIssue extends PartIssue {

    FullIssue(PartRequest request) {
        super(request);
    }

    @Override
    public String getSummary() {
        return "Wydanie pełne - liczba pozycji zamówienia: " + request.getItems().size();
    }

    @Override
    public void execute() {
        if (cancelIfOrderClosed()) {
            return;
        }
        for (PartItem item : request.getItems()) {
            item.getPart().decreaseStock(item.getQuantity());   // krok 10
        }
        request.markFulfilled();                                 // krok 11
    }
}

import java.util.Map;

// Produkt konkretny - wydanie dostępnej ilości, brakująca ilość jest usuwana z zamówienia
public class PartialIssue extends PartIssue {
    private final Map<PartItem, Integer> missingQuantities;

    PartialIssue(PartRequest request, Map<PartItem, Integer> missingQuantities) {
        super(request);
        this.missingQuantities = missingQuantities;
    }

    @Override
    public String getSummary() {
        StringBuilder summary = new StringBuilder("Wydanie częściowe:");
        for (Map.Entry<PartItem, Integer> entry : missingQuantities.entrySet()) {
            PartItem item = entry.getKey();
            summary.append(" ").append(item.getPart().getName())
                    .append(" - brakuje ").append(entry.getValue()).append(" szt.;");
        }
        return summary.append(" brakująca ilość zostanie usunięta z zamówienia,")
                .append(" mechanik musi zamówić ją ponownie").toString();
    }

    @Override
    public void execute() {
        if (cancelIfOrderClosed()) {
            return;
        }
        for (Map.Entry<PartItem, Integer> entry : missingQuantities.entrySet()) {
            entry.getKey().reduceQuantity(entry.getValue());
        }
        for (PartItem item : request.getItems()) {
            item.getPart().decreaseStock(item.getQuantity());
        }
        request.markFulfilled();
    }
}

// Produkt konkretny - części nie można wydać, zamówienie czeka na dostawę
public class WaitingIssue extends PartIssue {

    WaitingIssue(PartRequest request) {
        super(request);
    }

    @Override
    public String getSummary() {
        return "Brak części w magazynie: zamówienie oczekuje,"
                + " magazynier może zamówić część u dostawcy";
    }

    // Przepływ alternatywny 6: stan magazynowy się nie zmienia
    @Override
    public void execute() {
        request.markWaiting();
        request.getRepairOrder().changeStatus(RepairOrderStatus.WAITING_FOR_PARTS, "Oczekiwanie na części");
    }
}

import java.util.ArrayList;
import java.util.List;

// Klient metody fabrykującej - działa tylko na typie PartIssue
public class WarehouseServiceImpl {
    private final List<PartRequest> partRequests = new ArrayList<>();

    public void addPartRequest(PartRequest request) {
        partRequests.add(request);
    }

    // Krok 7: podsumowanie wydania pokazywane magazynierowi przed potwierdzeniem
    public String getIssueSummary(int requestId) {
        return PartIssueFactory.createIssue(findRequest(requestId)).getSummary();
    }

    // Wydanie części: pełne, częściowe albo oznaczenie zamówienia jako oczekującego
    public void issueParts(int requestId) {
        PartIssue issue = PartIssueFactory.createIssue(findRequest(requestId));
        issue.execute();
    }

    private PartRequest findRequest(int requestId) {
        for (PartRequest request : partRequests) {
            if (request.getId() == requestId) {
                return request;
            }
        }
        throw new IllegalArgumentException("Nie ma zamówienia części o numerze " + requestId);
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

// Zamówienie części złożone przez mechanika
public class PartRequest {
    private static int nextId = 1;

    private final int id;
    private final RepairOrder repairOrder;
    private final List<PartItem> items = new ArrayList<>();
    private PartRequestStatus status;

    public PartRequest(RepairOrder repairOrder) {
        this.id = nextId++;
        this.repairOrder = repairOrder;
        this.status = PartRequestStatus.PENDING;
    }

    public void addItem(SparePart part, int quantity) {
        items.add(new PartItem(part, quantity));
    }

    public void markWaiting() { status = PartRequestStatus.WAITING; }

    public void markFulfilled() { status = PartRequestStatus.FULFILLED; }

    public void cancel() { status = PartRequestStatus.CANCELLED; }

    public int getId() { return id; }

    public RepairOrder getRepairOrder() { return repairOrder; }

    public List<PartItem> getItems() { return Collections.unmodifiableList(items); }

    public PartRequestStatus getStatus() { return status; }
}

// Pozycja zamówienia: część i zamawiana ilość
public class PartItem {
    private final SparePart part;
    private int quantity;

    public PartItem(SparePart part, int quantity) {
        this.part = part;
        this.quantity = quantity;
    }

    // Częściowe wydanie: brakująca ilość jest usuwana z zamówienia
    public void reduceQuantity(int quantity) {
        if (quantity <= 0 || quantity >= this.quantity) {
            throw new IllegalArgumentException("Niepoprawna ilość do usunięcia: " + quantity);
        }
        this.quantity -= quantity;
    }

    public SparePart getPart() { return part; }

    public int getQuantity() { return quantity; }
}

// Część zamienna w magazynie (uproszczona)
public class SparePart {
    private final String catalogNumber;
    private final String name;
    private int stockQuantity;
    private int reservedQuantity;   // ilość zarezerwowana dla zleceń (rezerwację pominięto w przykładzie)

    public SparePart(String catalogNumber, String name, int stockQuantity) {
        this.catalogNumber = catalogNumber;
        this.name = name;
        this.stockQuantity = stockQuantity;
    }

    // Wolna ilość: stan magazynowy minus ilość zarezerwowana dla zleceń
    public int getAvailableQuantity() {
        return stockQuantity - reservedQuantity;
    }

    public boolean isAvailable(int quantity) {
        return getAvailableQuantity() >= quantity;
    }

    public void decreaseStock(int quantity) {
        if (!isAvailable(quantity)) {
            throw new IllegalStateException("Za mało części " + name + " w magazynie");
        }
        stockQuantity -= quantity;
    }

    public String getName() { return name; }

    public int getStockQuantity() { return stockQuantity; }
}

// Zlecenie naprawy (uproszczone - numer i status)
public class RepairOrder {
    private final int id;
    private RepairOrderStatus status;

    public RepairOrder(int id, RepairOrderStatus status) {
        this.id = id;
        this.status = status;
    }

    public void changeStatus(RepairOrderStatus status, String comment) {
        this.status = status;
    }

    public int getId() { return id; }

    public RepairOrderStatus getStatus() { return status; }
}

public enum PartRequestStatus {
    PENDING,
    WAITING,
    READY_FOR_ISSUE,
    FULFILLED,
    CANCELLED
}

public enum RepairOrderStatus {
    ACCEPTED,
    IN_PROGRESS,
    WAITING_FOR_PARTS,
    WAITING_FOR_ESTIMATE_APPROVAL,
    COMPLETED,
    SETTLED
}

import static org.junit.jupiter.api.Assertions.*;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

public class PartIssueFactoryTest {
    private RepairOrder order;
    private SparePart oilFilter;
    private SparePart brakePads;
    private PartRequest request;

    @BeforeEach
    public void setUp() {
        order = new RepairOrder(15, RepairOrderStatus.IN_PROGRESS);
        oilFilter = new SparePart("FO-101", "Filtr oleju", 10);
        brakePads = new SparePart("KL-10", "Klocki hamulcowe", 10);
        request = new PartRequest(order);
        request.addItem(oilFilter, 1);
        request.addItem(brakePads, 4);
    }

    @Test
    public void testFullIssueWhenAllPartsAvailable() {
        PartIssue issue = PartIssueFactory.createIssue(request);
        assertInstanceOf(FullIssue.class, issue);

        issue.execute();

        assertEquals(9, oilFilter.getStockQuantity());
        assertEquals(6, brakePads.getStockQuantity());
        assertEquals(PartRequestStatus.FULFILLED, request.getStatus());
    }

    @Test
    public void testPartialIssueRemovesMissingQuantity() {
        brakePads.decreaseStock(7);   // w magazynie zostały 3 komplety klocków, zamówiono 4

        PartIssue issue = PartIssueFactory.createIssue(request);
        assertInstanceOf(PartialIssue.class, issue);
        assertTrue(issue.getSummary().contains("mechanik musi zamówić ją ponownie"));

        issue.execute();

        assertEquals(3, request.getItems().get(1).getQuantity(), "Brakująca ilość usunięta z zamówienia");
        assertEquals(0, brakePads.getStockQuantity());
        assertEquals(9, oilFilter.getStockQuantity());
        assertEquals(PartRequestStatus.FULFILLED, request.getStatus());
    }

    @Test
    public void testWaitingIssueWhenPartUnavailable() {
        brakePads.decreaseStock(10);   // brak klocków w magazynie

        PartIssue issue = PartIssueFactory.createIssue(request);
        assertInstanceOf(WaitingIssue.class, issue);

        issue.execute();

        assertEquals(PartRequestStatus.WAITING, request.getStatus());
        assertEquals(RepairOrderStatus.WAITING_FOR_PARTS, order.getStatus());
        assertEquals(10, oilFilter.getStockQuantity(), "Stan magazynowy bez zmian");
    }

    @Test
    public void testRequestCancelledWhenOrderClosedInMeantime() {
        PartIssue issue = PartIssueFactory.createIssue(request);
        order.changeStatus(RepairOrderStatus.COMPLETED, "Naprawa zakończona");

        issue.execute();

        assertEquals(PartRequestStatus.CANCELLED, request.getStatus());
        assertEquals(10, oilFilter.getStockQuantity(), "Stan magazynowy bez zmian");
        assertEquals(10, brakePads.getStockQuantity());
    }

    @Test
    public void testWarehouseServiceUsesFactory() {
        WarehouseServiceImpl warehouse = new WarehouseServiceImpl();
        warehouse.addPartRequest(request);

        assertTrue(warehouse.getIssueSummary(request.getId()).startsWith("Wydanie pełne"));
        warehouse.issueParts(request.getId());

        assertEquals(PartRequestStatus.FULFILLED, request.getStatus());
    }
