Metoda fabrykująca

Kacper Bednarz

Problem

W przypadku użycia „Wydaj części mechanikowi” system sprawdza dostępność zamówionych części (krok 6) i postępuje na jeden z trzech sposobów:

  • wszystkie części są dostępne – wydanie pełne: stan magazynowy jest pomniejszany, a zamówienie oznaczane jako zrealizowane (kroki 10–11);
  • części jest mniej, niż zamówiono – wydanie częściowe: brakująca ilość jest usuwana z zamówienia, a mechanik musi zamówić ją ponownie (przepływ alternatywny 6);
  • części nie ma w magazynie – nic nie jest wydawane, zamówienie czeka, a zlecenie dostaje status „oczekuje na części” (przepływ alternatywny 6).

Gdyby serwis magazynu sam rozróżniał te przypadki rozbudowanymi instrukcjami warunkowymi, logika wydania byłaby w jednym dużym miejscu, a dodanie nowego sposobu wydania wymagałoby zmiany istniejącego kodu.

Rozwiązanie

Metoda fabrykująca tworzy obiekt, ale zwalnia kod, który z niego korzysta, z decyzji, jaki konkretny obiekt ma powstać. Tutaj o rodzaju wydania decyduje stan magazynu.

  • PartIssue – produkt abstrakcyjny z metodą execute(), która zapisuje skutki wydania.
  • FullIssue, PartialIssue, WaitingIssue – produkty konkretne, po jednym dla każdego sposobu ze scenariusza.
  • PartIssueFactory.createIssue() – metoda fabrykująca. Porównuje stan magazynowy każdej części z zamówioną ilością i zwraca odpowiedni produkt. Brak którejkolwiek części oznacza, że całe zamówienie czeka.

Z fabryki korzysta metoda WarehouseService.issueParts() z diagramu klas: tworzy wydanie przez createIssue() i wywołuje execute(), nie wiedząc, który rodzaj wydania powstał. Wydanie częściowe korzysta z metody PartItem.reduceQuantity() z diagramu klas. Struktura odpowiada schematowi metody fabrykującej z instrukcji: fabryka, produkt abstrakcyjny i produkty konkretne. Klasy SparePart, PartRequest i RepairOrder są w kodzie uproszczone do pól potrzebnych przy wydaniu.

Kod wzorca
// Fabryka z metodą fabrykującą
public class PartIssueFactory {

    // Na podstawie stanu magazynu wybiera, jaki rodzaj wydania utworzyć
    public static PartIssue createIssue(PartRequest request) {
        boolean missingParts = false;
        for (PartItem item : request.getItems()) {
            int stock = item.getPart().getStockQuantity();
            // Brak części w magazynie - całe zamówienie czeka
            if (stock == 0) {
                return new WaitingIssue(request);
            }
            if (stock < item.getQuantity()) {
                missingParts = true;
            }
        }
        if (missingParts) {
            return new PartialIssue(request);
        }
        return new FullIssue(request);
    }
}

// Produkt abstrakcyjny - sposób wydania części
public abstract class PartIssue {
    protected final PartRequest request;

    protected PartIssue(PartRequest request) {
        this.request = request;
    }

    // Zapisuje w systemie skutki wydania
    public abstract void execute();
}

// Produkt konkretny - wszystkie części są dostępne
public class FullIssue extends PartIssue {

    public FullIssue(PartRequest request) {
        super(request);
    }

    @Override
    public void execute() {
        // Pomniejszenie stanu magazynowego o zamówione ilości
        for (PartItem item : request.getItems()) {
            item.getPart().decreaseStock(item.getQuantity());
        }
        request.markFulfilled();
    }
}

// Produkt konkretny - części jest mniej, niż zamówiono
public class PartialIssue extends PartIssue {

    public PartialIssue(PartRequest request) {
        super(request);
    }

    @Override
    public void execute() {
        for (PartItem item : request.getItems()) {
            int stock = item.getPart().getStockQuantity();
            // Brakująca ilość jest usuwana z zamówienia - mechanik musi zamówić ją ponownie
            if (stock < item.getQuantity()) {
                item.reduceQuantity(item.getQuantity() - stock);
            }
            item.getPart().decreaseStock(item.getQuantity());
        }
        request.markFulfilled();
    }
}

// Produkt konkretny - części nie ma w magazynie
public class WaitingIssue extends PartIssue {

    public WaitingIssue(PartRequest request) {
        super(request);
    }

    @Override
    public void execute() {
        // Nic nie jest wydawane - zamówienie i zlecenie czekają na części
        request.markWaiting();
        request.getRepairOrder().changeStatus(RepairOrderStatus.WAITING_FOR_PARTS);
    }
}

import java.util.ArrayList;
import java.util.List;

// Zamówienie części złożone przez mechanika
public class PartRequest {
    private final RepairOrder repairOrder;
    private final List<PartItem> items = new ArrayList<>();
    private PartRequestStatus status = PartRequestStatus.PENDING;

    public PartRequest(RepairOrder repairOrder) {
        this.repairOrder = repairOrder;
    }

    public void addItem(SparePart part, int quantity) {
        items.add(new PartItem(part, quantity));
    }

    public void markWaiting() { status = PartRequestStatus.WAITING; }

    public void markFulfilled() { status = PartRequestStatus.FULFILLED; }

    public RepairOrder getRepairOrder() { return repairOrder; }

    public List<PartItem> getItems() { return items; }

    public PartRequestStatus getStatus() { return status; }
}

// Pozycja zamówienia: część i zamawiana ilość
public class PartItem {
    private final SparePart part;
    private int quantity;

    public PartItem(SparePart part, int quantity) {
        this.part = part;
        this.quantity = quantity;
    }

    // Usuwa z pozycji brakującą ilość
    public void reduceQuantity(int quantity) {
        this.quantity -= quantity;
    }

    public SparePart getPart() { return part; }

    public int getQuantity() { return quantity; }
}

// Część zamienna w magazynie
public class SparePart {
    private final String name;
    private int stockQuantity;

    public SparePart(String name, int stockQuantity) {
        this.name = name;
        this.stockQuantity = stockQuantity;
    }

    public void decreaseStock(int quantity) {
        stockQuantity -= quantity;
    }

    public String getName() { return name; }

    public int getStockQuantity() { return stockQuantity; }
}

// Zlecenie naprawy (tylko status potrzebny przy wydaniu części)
public class RepairOrder {
    private RepairOrderStatus status = RepairOrderStatus.IN_PROGRESS;

    public void changeStatus(RepairOrderStatus status) {
        this.status = status;
    }

    public RepairOrderStatus getStatus() { return status; }
}

public enum PartRequestStatus {
    PENDING,
    WAITING,
    READY_FOR_ISSUE,
    FULFILLED,
    CANCELLED
}

public enum RepairOrderStatus {
    ACCEPTED,
    IN_PROGRESS,
    WAITING_FOR_PARTS,
    WAITING_FOR_ESTIMATE_APPROVAL,
    COMPLETED,
    SETTLED
}
Kod testowy wzorca
import static org.junit.jupiter.api.Assertions.*;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

public class PartIssueFactoryTest {
    private RepairOrder order;
    private SparePart oilFilter;
    private SparePart brakePads;
    private PartRequest request;

    @BeforeEach
    public void setUp() {
        order = new RepairOrder();
        oilFilter = new SparePart("Filtr oleju", 10);
        brakePads = new SparePart("Klocki hamulcowe", 10);
        request = new PartRequest(order);
        request.addItem(oilFilter, 1);
        request.addItem(brakePads, 4);
    }

    @Test
    public void testFullIssue() {
        PartIssue issue = PartIssueFactory.createIssue(request);
        assertInstanceOf(FullIssue.class, issue);

        issue.execute();

        assertEquals(9, oilFilter.getStockQuantity());
        assertEquals(6, brakePads.getStockQuantity());
        assertEquals(PartRequestStatus.FULFILLED, request.getStatus());
    }

    @Test
    public void testPartialIssue() {
        // W magazynie zostają 3 komplety klocków, a zamówiono 4
        brakePads.decreaseStock(7);

        PartIssue issue = PartIssueFactory.createIssue(request);
        assertInstanceOf(PartialIssue.class, issue);

        issue.execute();

        assertEquals(3, request.getItems().get(1).getQuantity());
        assertEquals(0, brakePads.getStockQuantity());
        assertEquals(PartRequestStatus.FULFILLED, request.getStatus());
    }

    @Test
    public void testWaitingIssue() {
        // Klocków nie ma w magazynie
        brakePads.decreaseStock(10);

        PartIssue issue = PartIssueFactory.createIssue(request);
        assertInstanceOf(WaitingIssue.class, issue);

        issue.execute();

        assertEquals(10, oilFilter.getStockQuantity());
        assertEquals(PartRequestStatus.WAITING, request.getStatus());
        assertEquals(RepairOrderStatus.WAITING_FOR_PARTS, order.getStatus());
    }
}
